apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: model-router-ingress
  annotations:
    cert-manager.io/cluster-issuer: "letsencrypt-prod"
    nginx.ingress.kubernetes.io/ssl-redirect: "true"
spec:
  tls:
  - hosts:
    - api.modelrouter.com
    secretName: modelrouter-tls
  rules:
  - host: api.modelrouter.com
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: model-router-service
            port:
              number: 80
